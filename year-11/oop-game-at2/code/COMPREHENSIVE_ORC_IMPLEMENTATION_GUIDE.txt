        super().__init__(x, y, frames)
        
        # Orc-specific attributes (stronger than regular enemies, slower than skeletons)

        self.aggro_range = 180  # Slightly less than skeleton (220)

    def update(self, player):
        """
        Update the Orc based on its state and the player.
        
        Args:
            player: The player object to interact with.
        """
        # For basic implementation, just use parent update logic
        super().update(player)
    
    def take_damage(self, amount):
        """
        Take damage.
        
        Args:
            amount (int): The amount of damage to take.
            
        Returns:
            bool: True if the Orc died, False otherwise.
        """
        # For basic implementation, just use parent damage handling
        return super().take_damage(amount)
    
    def draw(self, screen):
        """
        Draw the Orc with its current animation frame.
        
        Args:
            screen: The pygame surface to draw on.

        # For basic implementation, just use parent drawing
        super().draw(screen)

### Step 2: Add Orc Asset Loading to AssetLoader

Add the following method to `/utils/asset_loader.py` in the AssetLoader class:

```python
def get_orc_frames(self):
    """Get animation frames for Orc enemy."""
    frames = {
        "idle": [],
        "attack": [],
        "hurt": [],
        "death": []
    }
    
    # Get the project root directory
    current_dir = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
    
    try:
        # Try to load from Orc folder
        orc_path = os.path.join(current_dir, "assets", "Orc", "Orc")
        
        # Animation types and expected frame counts
        animations = {
            "idle": {"file": "Orc-Idle.png", "frames": 6},
            "attack": {"file": "Orc-Attack01.png", "frames": 7},
            "hurt": {"file": "Orc-Hurt.png", "frames": 4},
            "death": {"file": "Orc-Death.png", "frames": 8}
        }
        
        # Load each animation type
        for anim_type, info in animations.items():
            anim_path = os.path.join(orc_path, info["file"])
            
            if os.path.exists(anim_path):
                try:
                    sheet = pygame.image.load(anim_path).convert_alpha()
                    
                    # Extract frames from sprite sheet
                    frame_count = info["frames"]
                    frame_width = sheet.get_width() // frame_count
                    frame_height = sheet.get_height()
                    
                    # Create each frame
                    for i in range(frame_count):
                        frame_rect = pygame.Rect(i * frame_width, 0, frame_width, frame_height)
                        frame = pygame.Surface((frame_width, frame_height), pygame.SRCALPHA)
                        frame.blit(sheet, (0, 0), frame_rect)
                        
                        # Scale the frame to match enemy size (same as skeleton)
                        frame = pygame.transform.scale(frame, (90, 115))
                        frames[anim_type].append(frame)
                except Exception as e:
                    print(f"Failed to load Orc {anim_type} animation: {e}")
                    # Create fallback frame for this animation
                    if frames["idle"]:
                        frames[anim_type] = [frames["idle"][0]]
            else:
                print(f"Orc {anim_type} file not found: {anim_path}")
        
    except Exception as e:
        print(f"Could not load Orc assets: {e}")
        # Fall back to using skeleton frames if Orc assets fail
        frames = {
            "idle": self.get_enemy_frames("idle"),
            "attack": self.get_enemy_frames("attack"),
            "hurt": self.get_enemy_frames("hurt"),
            "death": self.get_enemy_frames("death")
        }
    
    # If any frame list is empty, use the idle frame as fallback
    for state in ["attack", "hurt", "death"]:
        if not frames[state] and frames["idle"]:
            frames[state] = frames["idle"]
    
    return frames
```

## Troubleshooting Guide

### Common Issues and Solutions

1. **ImportError for Orc class**
   - **Cause**: Missing import in `game.py`
   - **Solution**: Add `from characters.orc import Orc` to imports

2. **Orcs not spawning**
   - **Cause**: Only appear in Level 2 by default
   - **Solution**: Add temporary test Orc to Level 1 or advance to Level 2

3. **Orcs appear as multiple static images in a line**
   - **Cause**: Using outdated sprite loading method
   - **Solution**: Use the sprite sheet parsing method from Step 2

4. **Orcs disappear during level setup**
   - **Cause**: Enemy validation system not handling Orcs
   - **Solution**: Update validation code as shown in Step 6

5. **AttributeError: 'NoneType' object has no attribute...**
   - **Cause**: Asset loading failed and no fallback was used
   - **Solution**: Verify `get_enemy_frames("idle")` method exists in AssetLoader

6. **Game crashes when creating Orcs**
   - **Cause**: Missing `game_map` assignment
   - **Solution**: Ensure `orc.game_map = self.game_map` is in `create_orc` method
