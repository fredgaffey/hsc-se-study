BEGIN add(num) 
  SET ans = num(1) + num(2) 
  RETURN num(1)+ " + " +num(2) = ans 
END add(num) 


BEGIN subtract(num) 
  SET ans = num(1) - num(2) 
  RETURN num(1)+ " - " +num(2) = ans 
END subtract(num) 
 

BEGIN multiply(num) 
  SET multiply_ans = num(1) * num(2) 
  RETURN num(1)+ " x " +num(2) = ans 
END multiply(num) 
  

BEGIN division(num) 
  SET ans = num(1) /  num(2)  
  RETURN num(1)+”  ÷  “+num(2) = ans 
END division 
  

BEGIN menu 
  DISPLAY "Please insert a letter 
      Menu:

      s for subtraction 
      m for multiplication 
      d for division 
      q for quit" 
  
    	GET menu_choice 

  CASEWHERE:
